```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

import numpy as np
import torch

from common_pde import FIELD_DATA_DIR, GRID_NX, GRID_NY, compute_boundary_masks


@dataclass(frozen=True)
class CaseBatch:
    xy_norm: torch.Tensor
    xy_phys: torch.Tensor
    target: torch.Tensor
    domain: torch.Tensor
    boundary: dict[str, torch.Tensor]
    spans: dict[str, float]


def _normalize_xy(x: np.ndarray, y: np.ndarray) -> tuple[np.ndarray, dict[str, float]]:
    x_min = float(x.min())
    x_max = float(x.max())
    y_min = float(y.min())
    y_max = float(y.max())
    x_span = x_max - x_min
    y_span = y_max - y_min
    x_norm = 2.0 * (x - x_min) / (x_span + 1e-12) - 1.0
    y_norm = 2.0 * (y - y_min) / (y_span + 1e-12) - 1.0
    return np.column_stack([x_norm, y_norm]).astype(np.float32), {
        "x_min": x_min,
        "x_max": x_max,
        "y_min": y_min,
        "y_max": y_max,
        "x_span": x_span,
        "y_span": y_span,
    }


def load_case(case_id: int, device: torch.device, data_dir: Path = FIELD_DATA_DIR) -> CaseBatch:
    path = data_dir / f"case_{case_id}_full.npz"
    if not path.exists():
        raise FileNotFoundError(f"场数据不存在: {path}")

    data = np.load(path)
    x = data["x"].astype(np.float32)
    y = data["y"].astype(np.float32)
    target = np.column_stack([
        data["u"].astype(np.float32),
        data["v"].astype(np.float32),
        data["p"].astype(np.float32),
        data["T"].astype(np.float32),
    ])
    domain = data["domain"].astype(np.int64)
    xy_norm, spans = _normalize_xy(x, y)
    masks = compute_boundary_masks(x, y, domain, GRID_NX, GRID_NY)

    return CaseBatch(
        xy_norm=torch.from_numpy(xy_norm).to(device),
        xy_phys=torch.from_numpy(np.column_stack([x, y]).astype(np.float32)).to(device),
        target=torch.from_numpy(target).to(device),
        domain=torch.from_numpy(domain).to(device),
        boundary={name: torch.from_numpy(mask).to(device) for name, mask in masks.items()},
        spans=spans,
    )
```

```python
from __future__ import annotations

import torch


def _grad(value: torch.Tensor, xy_norm: torch.Tensor) -> torch.Tensor:
    return torch.autograd.grad(
        value.sum(),
        xy_norm,
        create_graph=True,
        retain_graph=True,
    )[0]


def first_derivatives(
    value: torch.Tensor,
    xy_norm: torch.Tensor,
    *,
    x_span: float,
    y_span: float,
) -> tuple[torch.Tensor, torch.Tensor]:
    grad_norm = _grad(value, xy_norm)
    # xy_norm = 2 * (xy - min) / span - 1，因此 d/dx = d/dx_norm * 2/span。
    return grad_norm[:, 0] * (2.0 / x_span), grad_norm[:, 1] * (2.0 / y_span)


def second_derivatives(
    first_dx: torch.Tensor,
    first_dy: torch.Tensor,
    xy_norm: torch.Tensor,
    *,
    x_span: float,
    y_span: float,
) -> tuple[torch.Tensor, torch.Tensor]:
    dxx_norm = _grad(first_dx, xy_norm)[:, 0]
    dyy_norm = _grad(first_dy, xy_norm)[:, 1]
    return dxx_norm * (2.0 / x_span), dyy_norm * (2.0 / y_span)
```

```python
from __future__ import annotations

import torch
import torch.nn as nn


class NavierStokesPINN(nn.Module):
    def __init__(self, hidden: int = 256, n_layers: int = 6):
        super().__init__()
        layers: list[nn.Module] = [nn.Linear(2, hidden), nn.Tanh()]
        for _ in range(n_layers - 1):
            layers.extend([nn.Linear(hidden, hidden), nn.Tanh()])
        layers.append(nn.Linear(hidden, 4))
        self.net = nn.Sequential(*layers)
        self._init_weights()

    def _init_weights(self) -> None:
        for module in self.modules():
            if isinstance(module, nn.Linear):
                nn.init.xavier_normal_(module.weight)
                nn.init.zeros_(module.bias)

    def forward(self, xy_norm: torch.Tensor) -> torch.Tensor:
        return self.net(xy_norm)
```

```python
from __future__ import annotations

import torch
import torch.nn.functional as F

from common_pde import CP_AIR, K_AIR, K_ALUM, MU_AIR, Q_HEAT, RHO_AIR, T_AMB, U_IN
from pde_pinn.data import CaseBatch
from pde_pinn.residuals import first_derivatives, second_derivatives


FIELD_SCALE = torch.tensor([5.0, 1.0, 25.0, 20.0])
FIELD_OFFSET = torch.tensor([0.0, 0.0, 0.0, 293.15])


def default_loss_weights(epoch: int) -> dict[str, float]:
    pde = min(1.0, max(0.0, (epoch - 200) / 800.0))
    return {
        "data": 1.0,
        "bc": 0.1,
        "continuity": 0.01 * pde,
        "momentum": 0.01 * pde,
        "energy": 0.01 * pde,
    }


def _physical_output(raw: torch.Tensor) -> torch.Tensor:
    scale = FIELD_SCALE.to(raw.device)
    offset = FIELD_OFFSET.to(raw.device)
    return raw * scale + offset


def compute_single_case_loss(
    model: torch.nn.Module,
    batch: CaseBatch,
    weights: dict[str, float],
) -> tuple[torch.Tensor, dict[str, float]]:
    xy_norm = batch.xy_norm.detach().clone().requires_grad_(True)
    pred = _physical_output(model(xy_norm))
    target = batch.target
    domain = batch.domain
    air = domain == 0
    solid = domain == 1

    scale = FIELD_SCALE.to(pred.device)
    data_loss = F.mse_loss((pred[air, :3] - target[air, :3]) / scale[:3], torch.zeros_like(pred[air, :3]))
    data_loss = data_loss + F.mse_loss((pred[:, 3] - target[:, 3]) / scale[3], torch.zeros_like(pred[:, 3]))

    u, v, p, temp = pred[:, 0], pred[:, 1], pred[:, 2], pred[:, 3]
    x_span = batch.spans["x_span"]
    y_span = batch.spans["y_span"]
    du_dx, du_dy = first_derivatives(u, xy_norm, x_span=x_span, y_span=y_span)
    dv_dx, dv_dy = first_derivatives(v, xy_norm, x_span=x_span, y_span=y_span)
    dp_dx, dp_dy = first_derivatives(p, xy_norm, x_span=x_span, y_span=y_span)
    dt_dx, dt_dy = first_derivatives(temp, xy_norm, x_span=x_span, y_span=y_span)
    d2u_dx2, d2u_dy2 = second_derivatives(du_dx, du_dy, xy_norm, x_span=x_span, y_span=y_span)
    d2v_dx2, d2v_dy2 = second_derivatives(dv_dx, dv_dy, xy_norm, x_span=x_span, y_span=y_span)
    d2t_dx2, d2t_dy2 = second_derivatives(dt_dx, dt_dy, xy_norm, x_span=x_span, y_span=y_span)

    continuity = F.mse_loss((du_dx + dv_dy)[air] / 500.0, torch.zeros_like(du_dx[air]))
    mx = RHO_AIR * (u * du_dx + v * du_dy) + dp_dx - MU_AIR * (d2u_dx2 + d2u_dy2)
    my = RHO_AIR * (u * dv_dx + v * dv_dy) + dp_dy - MU_AIR * (d2v_dx2 + d2v_dy2)
    momentum = F.mse_loss(mx[air] / 3000.0, torch.zeros_like(mx[air]))
    momentum = momentum + F.mse_loss(my[air] / 3000.0, torch.zeros_like(my[air]))
    energy_air = RHO_AIR * CP_AIR * (u * dt_dx + v * dt_dy) - K_AIR * (d2t_dx2 + d2t_dy2)
    energy_solid = K_ALUM * (d2t_dx2 + d2t_dy2) + Q_HEAT
    energy = F.mse_loss(energy_air[air] / 2.5e7, torch.zeros_like(energy_air[air]))
    if solid.any():
        energy = energy + F.mse_loss(energy_solid[solid] / Q_HEAT, torch.zeros_like(energy_solid[solid]))

    inlet = batch.boundary["inlet"]
    wall = batch.boundary["wall"]
    bc = F.mse_loss(u[inlet] / U_IN, torch.ones_like(u[inlet]))
    bc = bc + F.mse_loss(v[inlet] / U_IN, torch.zeros_like(v[inlet]))
    bc = bc + F.mse_loss((temp[inlet] - T_AMB) / 20.0, torch.zeros_like(temp[inlet]))
    if wall.any():
        bc = bc + F.mse_loss(u[wall] / U_IN, torch.zeros_like(u[wall]))
        bc = bc + F.mse_loss(v[wall] / U_IN, torch.zeros_like(v[wall]))

    total = (
        weights["data"] * data_loss
        + weights["bc"] * bc
        + weights["continuity"] * continuity
        + weights["momentum"] * momentum
        + weights["energy"] * energy
    )
    return total, {
        "data": float(data_loss.detach().cpu()),
        "bc": float(bc.detach().cpu()),
        "continuity": float(continuity.detach().cpu()),
        "momentum": float(momentum.detach().cpu()),
        "energy": float(energy.detach().cpu()),
        "total": float(total.detach().cpu()),
    }
```

```python
from __future__ import annotations

import math

import numpy as np
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score


def field_metrics(pred: np.ndarray, target: np.ndarray, domain: np.ndarray) -> dict[str, float]:
    metrics: dict[str, float] = {}
    air = domain == 0
    for index, name in enumerate(["u", "v", "p", "T"]):
        mask = air if name != "T" else np.ones(len(domain), dtype=bool)
        mse = mean_squared_error(target[mask, index], pred[mask, index])
        metrics[f"{name}_R2"] = float(r2_score(target[mask, index], pred[mask, index]))
        metrics[f"{name}_MAE"] = float(mean_absolute_error(target[mask, index], pred[mask, index]))
        metrics[f"{name}_RMSE"] = float(math.sqrt(mse))
    return metrics
```
